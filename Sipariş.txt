<script>
const BIN_ID = "696dfab7d0ea881f4075b9a8";
const MASTER_KEY = "$2a$10$ze2fKA7ejuDI/zkMVFwgA..2tG/eSJMfRn6QzlH85ktarzLTshMCe";
const PASSWORD = "1234";
const WA_PHONE = "905433553177";

const API_URL = "https://api.jsonbin.io/v3/b/" + BIN_ID;

const products=[
"Ayran","Çiğ Köfte","Dürüm",
"Marul","Salatalık","Mor Lahana","Havuç",
"Limon","Domates","Maydanoz","Roka","Nane"
];

function login(){
if(pass.value===PASSWORD){
loginBox.style.display="none";
app.style.display="block";
loadOrders();
}else alert("Şifre yanlış");
}

function addItem(){
const d=document.createElement("div");
d.className="item";
d.innerHTML=`
<select class="product">
<option value="">Ürün</option>
${products.map(p=>`<option>${p}</option>`).join("")}
</select>
<input type="number" class="qty" placeholder="Miktar" oninput="calc()">
<select class="unit" onchange="calc()">
<option>Kg</option><option>Adet</option>
</select>
<button style="background:#555;margin-top:5px" onclick="this.parentElement.remove();calc()">Sil</button>
`;
items.appendChild(d);
}
addItem();

function calc(){
let kg=0,adet=0;
document.querySelectorAll(".item").forEach(i=>{
const q=Number(i.querySelector(".qty").value);
if(!q) return;
i.querySelector(".unit").value==="Kg"?kg+=q:adet+=q;
});
kgTotal.innerText=kg;
adetTotal.innerText=adet;
}

function loadOrders(){
fetch(API_URL,{
headers:{"X-Master-Key":MASTER_KEY}
})
.then(r=>r.json())
.then(r=>{
const d=r.record;
orders.innerHTML=d.orders.map(o=>`
<div class="order">
<b>${o.name}</b><br>
${o.items}<br>
<small>${o.time} | Kg:${o.kg} Adet:${o.adet}</small>
</div>
`).join("");
});
}

function saveOrder(){
if(!customer.value.trim()) return alert("İsim yaz");
const list=[];
document.querySelectorAll(".item").forEach(i=>{
const p=i.querySelector(".product").value;
const q=i.querySelector(".qty").value;
const u=i.querySelector(".unit").value;
if(p&&q) list.push(`${p} ${q} ${u}`);
});
if(!list.length) return alert("Ürün gir");

fetch(API_URL,{
headers:{"X-Master-Key":MASTER_KEY}
})
.then(r=>r.json())
.then(r=>{
const d=r.record;
d.orders.push({
name:customer.value.trim(),
items:list.join(" | "),
kg:kgTotal.innerText,
adet:adetTotal.innerText,
time:new Date().toLocaleString()
});
fetch(API_URL,{
method:"PUT",
headers:{
"Content-Type":"application/json",
"X-Master-Key":MASTER_KEY
},
body:JSON.stringify(d)
}).then(()=>{
customer.value="";
items.innerHTML="";
addItem();
loadOrders();
});
});
}

function clearOrders(){
if(confirm("TÜM SİPARİŞLER SİLİNSİN Mİ?")){
fetch(API_URL,{
method:"PUT",
headers:{
"Content-Type":"application/json",
"X-Master-Key":MASTER_KEY
},
body:JSON.stringify({orders:[]})
}).then(()=>loadOrders());
}
}

function sendWA(){
fetch(API_URL,{
headers:{"X-Master-Key":MASTER_KEY}
})
.then(r=>r.json())
.then(r=>{
const d=r.record;
if(!d.orders.length) return alert("Sipariş yok");
let t="📦 SİPARİŞLER\n\n";
d.orders.forEach(o=>{
t+=`${o.name}\n${o.items}\nKg:${o.kg} Adet:${o.adet}\n${o.time}\n\n`;
});
window.open("https://wa.me/"+WA_PHONE+"?text="+encodeURIComponent(t),"_blank");
});
}
